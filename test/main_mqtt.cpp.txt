#include <Arduino.h>
#include <WiFi.h>
#include <PubSubClient.h>


#include "CommandManager.h"
#include "ValveManager.h"
#include "Valve.h"

Valve valves[] =
{
    Valve("Pelouse",25,1800),
    Valve("Potager",26,900),
    Valve("Serre",27,600)
};


ValveManager valveManager(
    valves,
    3
);


const char* ssid = "SFR_D8D8";

const char* password = "confinement";


const char* mqttServer =
    "192.168.1.10";



WiFiClient wifiClient;

PubSubClient mqtt(
    wifiClient
);



CommandManager commandManager(
    &mqtt,
    &valveManager
);



void setup()
{

    Serial.begin(115200);


    WiFi.begin(
        ssid,
        password
    );


    while(WiFi.status()!=WL_CONNECTED)
    {
        delay(500);
        Serial.print(".");
    }


    Serial.println("WiFi OK");



    mqtt.setServer(
        mqttServer,
        1883
    );


    commandManager.begin();

}




void loop()
{

    if(!mqtt.connected())
    {
        mqtt.connect(
            "esp32-arrosage"
        );
    }


    commandManager.update();


    valveManager.update();

}
