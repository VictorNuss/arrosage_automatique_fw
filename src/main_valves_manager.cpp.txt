#include <Arduino.h>

#include "Valve.h"
#include "ValveManager.h"



Valve valves[] =
{
    Valve("Pelouse",25,1800),
    Valve("Potager",26,900),
    Valve("Serre",27,600)
};



ValveManager valveManager(
    valves,
    sizeof(valves)/sizeof(valves[0])
);



void setup()
{
    Serial.begin(115200);
    valveManager.begin();
    // Test :
    // ouvre la vanne 2 pendant 30 secondes
    valveManager.open(
        1,
        30
    );

}



void loop()
{
    valveManager.update();


    static unsigned long lastPrint = 0;


    if(millis()-lastPrint > 5000)
    {
        valveManager.printStatus();

        lastPrint = millis();
    }
}
